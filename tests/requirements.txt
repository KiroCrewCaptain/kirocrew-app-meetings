# Test-only dependencies, pinned. Deliberately NOT at the repository root: a root
# requirements.txt makes a Kiro Crew registry install run pip into the gateway's
# interpreter (and is refused outright on the desktop build). At runtime the app
# needs only the stdlib and aiohttp, which the gateway already provides.
aiohttp==3.14.3
pytest==9.0.3
# The ported tests mark their coroutines `pytest.mark.asyncio` (strict mode). The
# 1.x line closes every loop it opens, which the `error` warning filter in
# pytest.ini needs: 0.21 left loops for the garbage collector.
pytest-asyncio==1.4.1.post1
flake8==7.5.0
# The IANA zone database the calendar parser resolves TZIDs against. Kiro Crew
# depends on it (setup.cfg: Windows has no system zone data, and newer Linux
# images moved the legacy link names out of theirs), so the gateway that runs
# this app always has it.
tzdata==2026.2
